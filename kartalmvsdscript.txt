local AntiBan = {

        task.wait(min + math.random() * (max - min))

    end,
    RandomBodyPart = function(self, char)
        return char:FindFirstChild(BODY[math.random(#BODY)]) or char:FindFirstChild("HumanoidRootPart")
    end
}

-- ===================== YARDIMCI FONKSİYONLAR =====================

local function isEnemy(model)
    if not model then return false end
    local plr = Players:GetPlayerFromCharacter(model)
    if not plr or plr == LocalPlayer then return false end
    return plr.Team ~= LocalPlayer.Team
end

local function isAlive(char)
    return char and char:FindFirstChild("Humanoid") and char.Humanoid.Health > 0
end

local function getClosestEnemy()
    local closest, closestDist = nil, math.huge
    for _, v in pairs(Players:GetPlayers()) do
        if v ~= LocalPlayer and isAlive(v.Character) and isEnemy(v.Character) then
            local hrp = v.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                if onScreen then
                    local dist = (Vector2.new(Mouse.X, Mouse.Y) - Vector2.new(pos.X, pos.Y)).Magnitude
                    if dist < closestDist then
                        closest = v
                        closestDist = dist
                    end
                end
            end
        end
    end
    return closest
end

local function getAllEnemies()
    local list = {}
    for _, v in pairs(Players:GetPlayers()) do
        if v ~= LocalPlayer and isAlive(v.Character) and isEnemy(v.Character) then
            table.insert(list, v)
        end
    end
    -- Karıştır
    for i = #list, 1, -1 do
        local j = math.random(1, i)
        list[i], list[j] = list[j], list[i]
    end
    return list
end

local function Shoot(char)
    if not ShootRemote or not char then return false end
    local part = AntiBan:RandomBodyPart(char)
    if not part then return false end
    ShootRemote:FireServer(AntiBan:RandomVector(), AntiBan:RandomVector(), part, AntiBan:RandomVector())
    return true
end

local function Stab(char)
    if not StabRemote or not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    StabRemote:FireServer(hrp)
    return true
end

local Hitbox = Window:AddSection("🎯 HITBOX")

_G.HitboxEnabled = false
_G.HitboxSize = 7
_G.HitboxTransparency = 0.6

local function HitboxLoop()
    while _G.HitboxEnabled do
        for _, v in pairs(Players:GetPlayers()) do
            if v ~= LocalPlayer and v.Character then
                local hrp = v.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    hrp.Size = Vector3.new(_G.HitboxSize, _G.HitboxSize, _G.HitboxSize)
                    hrp.Transparency = _G.HitboxTransparency
                    hrp.Material = Enum.Material.Neon
                    hrp.Color = Color3.fromRGB(255, 0, 0)
                end
                for _, part in ipairs(v.Character:GetDescendants()) do
                    if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                        part.Size = Vector3.new(_G.HitboxSize * 0.5, _G.HitboxSize * 0.5, _G.HitboxSize * 0.5)
                    end
                end
            end
        end
        task.wait(0.2)
    end
    -- Reset
    for _, v in pairs(Players:GetPlayers()) do
        if v ~= LocalPlayer and v.Character then
            local hrp = v.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Size = Vector3.new(2, 2, 1)
                hrp.Transparency = 1
                hrp.Material = Enum.Material.Plastic
            end
        end
    end
end

Hitbox:AddToggle("✦ Hitbox Expander", function(state)
    _G.HitboxEnabled = state
    if state then coroutine.wrap(HitboxLoop)() end
end)

Hitbox:AddSlider("✦ Size", 1, 7, 12, function(v) _G.HitboxSize = v end, true)

_G.ESP = false
Visuals:AddToggle("✦ ESP Box + Health", function(state)

        while _G.ESP do
            for _, v in pairs(Players:GetPlayers()) do
                if v ~= LocalPlayer and v.Character then
                    local hrp = v.Character:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        local box = hrp:FindFirstChild("ESP_Box")
                        if not box then
                            box = Instance.new("BoxHandleAdornment")
                            box.Name = "ESP_Box"
                            box.Adornee = hrp
                            box.AlwaysOnTop = true
                            box.ZIndex = 10
                            box.Parent = hrp
                        end
                        box.Size = Vector3.new(4, 6, 4)
                        box.Color3 = v.Team == LocalPlayer.Team and Color3.fromRGB(0,255,0) or Color3.fromRGB(255,50,50)
                        box.Transparency = 0.4
                        box.Visible = true
                        
                        -- Health bar
                        local billboard = hrp:FindFirstChild("ESP_Health")
                        if not billboard and v.Character:FindFirstChild("Humanoid") then
                            billboard = Instance.new("BillboardGui")
                            billboard.Name = "ESP_Health"
                            billboard.Size = UDim2.new(0, 100, 0, 20)
                            billboard.StudsOffset = Vector3.new(0, 3.5, 0)
                            billboard.AlwaysOnTop = true
                            local label = Instance.new("TextLabel", billboard)
                            label.Size = UDim2.new(1, 0, 1, 0)
                            label.BackgroundTransparency = 1
                            label.TextScaled = true
                            label.Font = Enum.Font.GothamBold
                            label.TextColor3 = Color3.fromRGB(255, 255, 255)
                            label.TextStrokeTransparency = 0.3
                            billboard.Parent = hrp
                        end
                        if billboard and billboard:IsA("BillboardGui") then
                            local label = billboard:FindFirstChildOfClass("TextLabel")
                            if label and v.Character:FindFirstChild("Humanoid") then
                                label.Text = v.Name .. " [" .. math.floor(v.Character.Humanoid.Health) .. "HP]"
                                label.TextColor3 = v.Character.Humanoid.Health > 50 and Color3.fromRGB(0,255,0) 
                                    or v.Character.Humanoid.Health > 25 and Color3.fromRGB(255,200,0) 
                                    or Color3.fromRGB(255,0,0)
                            end
                        end
                    end
                end
            end
            task.wait(0.3)
        end
        -- Cleanup
        for _, v in pairs(Players:GetPlayers()) do
            if v.Character then
                local hrp = v.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local box = hrp:FindFirstChild("ESP_Box")
                    if box then box:Destroy() end
                    local hp = hrp:FindFirstChild("ESP_Health")
                    if hp then hp:Destroy() end
                end
            end
        end
    end)()

local flyBodyVelocity = nil
Movement:AddToggle("✦ Fly", function(state)
    _G.Fly = state
    if state then
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            flyBodyVelocity = Instance.new("BodyVelocity")
            flyBodyVelocity.MaxForce = Vector3.new(1, 1, 1) * math.huge
            flyBodyVelocity.Velocity = Vector3.new(0, 0, 0)
            flyBodyVelocity.Parent = char.HumanoidRootPart
            
            coroutine.wrap(function()
                while _G.Fly do
                    local moveDir = Vector3.new()
                    if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + Camera.CFrame.LookVector end
                    if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - Camera.CFrame.LookVector end
                    if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - Camera.CFrame.RightVector end
                    if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + Camera.CFrame.RightVector end
                    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                    if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end
                    
                    if moveDir.Magnitude > 0 then
                        flyBodyVelocity.Velocity = moveDir.Unit * _G.FlySpeed
                    else
                        flyBodyVelocity.Velocity = Vector3.new(0, 0, 0)
                    end
                    task.wait(0.01)
                end
            end)()
        end
    else
        if flyBodyVelocity then
            flyBodyVelocity:Destroy()
            flyBodyVelocity = nil
        end
    end
end)
